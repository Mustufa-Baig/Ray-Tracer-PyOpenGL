#version 330 core

in vec3 fragmentColor;

out vec4 color;

uniform vec2 u_resolution;

struct Sphere {
	vec3 center;
	float radius;
	vec3 materialColor;
	float reflectivity;
};

struct Plane {
	vec3 normal;
	float d;
};

float hitSphere(vec3 center, float radius, vec3 ro, vec3 rd) {
    vec3 oc = ro - center;
    
    float b = dot(oc, rd);
    float c = dot(oc, oc) - radius * radius;
    float discriminant = b * b - c;
    
    if (discriminant > 0.0) {
        return -b - sqrt(discriminant);
    }    
    return -1.0;
}

float hitPlane(Plane p, vec3 ro, vec3 rd) {
    float denom = dot(rd, p.normal);
    if (denom < -0.0001) { 
        float t = -(dot(ro, p.normal) + p.d) / denom;
        if (t > 0.0) return t;
    }
    return -1.0;
}


void main()
{
	vec2 uv= gl_FragCoord.xy / u_resolution.xy;
	uv=(uv*2.0) -1.0;
	uv.x *= u_resolution.x/u_resolution.y;


	vec3 cam_pos = vec3(0.0, 0.0, -3.0); 
	vec3 cam_dir = normalize(vec3(uv, 1.0)); 


    const int NUM_SPHERES = 3;
    Sphere spheres[NUM_SPHERES];
    
    spheres[2] = Sphere(vec3(-2.0, 0.0, -1.0), 1.0, vec3(0.0, 0.5, 1.0),0.85);
    spheres[0] = Sphere(vec3(0.0, 0.0, 0.0), 1.0, vec3(1.0, 0.0, 0.0),0.2); 
    spheres[1] = Sphere(vec3(2.0, 0.0, 1.0), 1.0, vec3(0.0, 1.0, 0.0),0.7); 

    Plane floorPlane = Plane(vec3(0.0, 1.0, 0.0), 1.0);

    vec3 finalColor = vec3(0.0);
    vec3 reflectionMultiplier = vec3(1.0);
    const int MAX_BOUNCES=3;

    for (int bounce = 0; bounce < MAX_BOUNCES; bounce++) {
	    float closest_t = 999999.0; 
	    int hit_type = 0;
	    int hit_index = -1;         

	    for (int i = 0; i < NUM_SPHERES; i++) {
	        float t = hitSphere(spheres[i].center, spheres[i].radius, cam_pos, cam_dir);
	        if (t > 0.0 && t < closest_t) {
	            closest_t = t;
	            hit_type=1;
	            hit_index = i;
	        }
	    }

		float t_plane = hitPlane(floorPlane, cam_pos, cam_dir);
		if (t_plane > 0.0 && t_plane < closest_t) {
			closest_t = t_plane;
			hit_type = 2; 
		}

		if (hit_type == 0) {
			finalColor += reflectionMultiplier * vec3(0.1, 0.1, 0.1);
            break;
		}
		vec3 hitPoint = cam_pos + cam_dir * closest_t;
        vec3 normal;
        vec3 baseColor;
        float currentReflectivity;

        if (hit_type == 1) {
            normal = normalize(hitPoint - spheres[hit_index].center);
            baseColor = spheres[hit_index].materialColor;
            currentReflectivity = spheres[hit_index].reflectivity;
        } else {
            normal = floorPlane.normal;
            float checker = mod(floor(hitPoint.x) + floor(hitPoint.z), 2.0);
            baseColor = mix(vec3(0.2), vec3(0.7), checker); 
            currentReflectivity = 0.3; 
        }

	    vec3 lightDir = normalize(vec3(1.0, -1.0, 1.0));
        float diffuse = max(dot(normal, -lightDir), 0.0);
        vec3 directLight = baseColor * diffuse + (baseColor * 0.15);
        
        finalColor += reflectionMultiplier * directLight * (1.0 - currentReflectivity);
        reflectionMultiplier *= currentReflectivity;
        
        cam_pos = hitPoint + normal * 0.001; 
        cam_dir = reflect(cam_dir, normal); 
	}
	color = vec4(finalColor, 1.0);
}
