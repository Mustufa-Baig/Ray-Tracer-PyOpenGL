#version 330 core

in vec3 fragmentColor;
out vec4 color;

uniform vec2 u_resolution; 
uniform float u_frameCount; 
uniform sampler2D u_accumTexture; 

uniform vec3 u_camPos;
uniform vec3 u_camForward;
uniform vec3 u_camRight;
uniform vec3 u_camUp;

float seed; 

float random() {
    seed = fract(sin(seed) * 43758.5453123);
    return seed;
}

vec3 randomUnitVector() {
    float z = random() * 2.0 - 1.0;
    float a = random() * 6.28318530718;
    float r = sqrt(1.0 - z * z);
    return vec3(r * cos(a), r * sin(a), z);
}


struct Sphere {
    vec3 center;
    float radius;
    vec3 materialColor;
    float roughness; 
};

struct Plane {
    vec3 normal;
    float d; 
};


float hitSphere(vec3 center, float radius, vec3 ro, vec3 rd) {
    vec3 oc = ro - center;
    float b = dot(oc, rd);
    float c = dot(oc, oc) - radius * radius;
    float discriminant = b * b - c;
    if (discriminant > 0.0) return -b - sqrt(discriminant);
    return -1.0;
}

float hitPlane(Plane p, vec3 ro, vec3 rd) {
    float denom = dot(rd, p.normal);
    if (denom < -0.0001) { 
        float t = -(dot(ro, p.normal) + p.d) / denom;
        if (t > 0.0) return t;
    }
    return -1.0;
}

void main()
{
	float pixelHash = fract(sin(dot(gl_FragCoord.xy, vec2(12.9898, 78.233))) * 43758.5453123);
    
    seed = fract(pixelHash + u_frameCount * 0.61803398875);

    
    const int NUM_SPHERES = 3;
    Sphere spheres[NUM_SPHERES];
	spheres[2] = Sphere(vec3(-2.0, 0.0, -1.0), 1.0, vec3(0.0, 0.5, 1.0),0.1);
	spheres[0] = Sphere(vec3(0.0, 0.0, 0.0), 1.0, vec3(1.0, 1.0, 1.0),0.8);
	spheres[1] = Sphere(vec3(2.0, 0.0, 1.0), 1.0, vec3(0.0, 1.0, 0.0),0.2);

    Plane floorPlane = Plane(vec3(0.0, 1.0, 0.0), 1.0); 

    
    const int NUM_SAMPLES = 2; 
    vec3 totalSampleColor = vec3(0.0);

    for (int s = 0; s < NUM_SAMPLES; s++) {
        
        
        vec2 jitter = vec2(random() - 0.5, random() - 0.5);
        vec2 uv = (gl_FragCoord.xy + jitter) / u_resolution.xy;
        uv = uv * 2.0 - 1.0;
        uv.x *= u_resolution.x / u_resolution.y;

        vec3 cam_pos = u_camPos; 
        vec3 cam_dir = normalize(uv.x * u_camRight + uv.y * u_camUp + u_camForward);

        vec3 finalColor = vec3(0.0);
        vec3 reflectionMultiplier = vec3(1.0); 
        const int MAX_BOUNCES = 15;

        for (int bounce = 0; bounce < MAX_BOUNCES; bounce++) {
            
            float closest_t = 999999.0;
            int hit_type = 0; 
            int hit_index = -1;

            
            for (int i = 0; i < NUM_SPHERES; i++) {
                float t = hitSphere(spheres[i].center, spheres[i].radius, cam_pos, cam_dir);
                if (t > 0.0 && t < closest_t) {
                    closest_t = t;
                    hit_type = 1;
                    hit_index = i;
                }
            }

            
            float t_plane = hitPlane(floorPlane, cam_pos, cam_dir);
            if (t_plane > 0.0 && t_plane < closest_t) {
                closest_t = t_plane;
                hit_type = 2; 
            }

            
            if (hit_type == 0) {
                finalColor += reflectionMultiplier * vec3(0.8, 0.8, 0.8);
                break; 
            }

            
            vec3 hitPoint = cam_pos + cam_dir * closest_t;
            vec3 normal;
            vec3 baseColor;
            float currentRoughness;

            if (hit_type == 1) {
                normal = normalize(hitPoint - spheres[hit_index].center);
                baseColor = spheres[hit_index].materialColor;
                currentRoughness = spheres[hit_index].roughness;
            } else {
                normal = floorPlane.normal;
                float checker = mod(floor(hitPoint.x) + floor(hitPoint.z), 2.0);
                baseColor = mix(vec3(0.2), vec3(0.7), checker); 
                currentRoughness = 0.5;
            }

            
            reflectionMultiplier *= baseColor; 
            
            
            cam_pos = hitPoint + normal * 0.001; 
            
            
            vec3 pureReflection = reflect(cam_dir, normal);
            cam_dir = normalize(pureReflection + randomUnitVector() * currentRoughness);
            
            
            if (dot(cam_dir, normal) < 0.0) {
                cam_dir = normalize(cam_dir + normal);
            }
        }
        
        
        totalSampleColor += finalColor;
    }

    
    vec3 currentFrameAvg = totalSampleColor / float(NUM_SAMPLES);

    
    vec2 texCoord = gl_FragCoord.xy / u_resolution.xy;
    vec3 previousColor = texture(u_accumTexture, texCoord).rgb;
    
    
    vec3 blendedColor = mix(previousColor, currentFrameAvg, 1.0 / (u_frameCount + 1.0));

    color = vec4(blendedColor, 1.0);
}
